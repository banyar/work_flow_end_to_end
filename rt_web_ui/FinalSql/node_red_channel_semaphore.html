  • Slot ကို attempt တစ်ကြိမ်ချင်းမှာ ယူပါတယ် (retry loop တစ်ခုလုံးမှာ မဟုတ်)။ Retry ကြား time.Sleep မှာ slot ကို လွှတ်ထားလို့ တခြား ticket တွေ ဆက်သုံးလို့ ရပါတယ်။
  • Slot စောင့်တာကြောင့် retry job ထဲ မပို့ပါ။ Ticket တိုင်း slot ရတဲ့အထိ စောင့်ပြီး Node-RED ကို ခေါ်ပါတယ်။ Retry job ထဲ ရောက်တာက ယခင်အတိုင်း Node-RED call တွေ attempt ကုန်အောင် fail ဖြစ်မှပဲ ဖြစ်ပါတယ် (RETRY_WORKER_ENABLED=false ဆိုရင် FAILED_PERMANENT)။
  • အကြာဆုံး ဘယ်လောက် စောင့်ရမလဲ: ကိုယ့်ရှေ့မှာ စောင့်နေသူ အရေအတွက် ÷ 8 × call ကြာချိန်ပါ။ ဥပမာ ticket 100 ခု burst ဖြစ်ရင် နောက်ဆုံး ticket က (100 ÷ 8) × 5s ≈ 60 စက္ကန့် စောင့်ရပါတယ်။
  • Deadlock မဖြစ်ပါ: slot ကိုင်ထားတဲ့ call တိုင်းက NODE_RED_TIMEOUT (HTTP timeout) နဲ့ အဆုံးသတ်ပြီး defer နဲ့ slot ပြန်လွှတ်လို့ Node-RED မဖြေတောင် slot တွေ အမြဲ ပြန်လွတ်ပါတယ်။
  • defer release() ဖြစ်လို့ error / panic ဖြစ်လည်း slot ပြန်လွှတ်ပါတယ်။ Slot ပျောက် (leak) မဖြစ်ပါ။

4.1 Struct

type nodeRedClientImpl struct {
	baseURL      string
	apiKey       string
	environment  string
	maxRetries   int
	retryDelayMs time.Duration
	httpClient   *http.Client
	slots chan struct{} // make(chan struct{}, NODE_RED_MAX_CONCURRENCY); nil = no limit
}

4.2 Slot ယူ / လွှတ်

// acquire takes a slot, waiting as long as it takes for one to free up.
// Every holder gives its slot back within NODE_RED_TIMEOUT (defer release),
// so the wait always ends.
func (c *nodeRedClientImpl) acquire() (release func()) {
	if c.slots == nil { // NODE_RED_MAX_CONCURRENCY=0: no limit (rollback switch)
		return func() {}
	}
	c.slots <- struct{}{} // blocks while 8/8 are in use
	return func() { <-c.slots }
}

// fetchLimited is one Node-RED call inside a slot.
func (c *nodeRedClientImpl) fetchLimited(cpeID string) (*models.NodeRedResponse, error) {
	release := c.acquire()
	defer release()
	return c.fetchCPEStatusSingle(cpeID)
}

4.3 Retry loop (FetchCPEStatus)

for attempt := 1; attempt <= c.maxRetries; attempt++ {
	response, err := c.fetchLimited(cpeID)             // was: c.fetchCPEStatusSingle(cpeID)
	if err == nil {
		return response, nil
	}
	if errors.Is(err, common.ERR_CPE_NOT_FOUND) {
		return nil, err
	}
	lastErr = err
	if attempt < c.maxRetries {
		time.Sleep(c.retryDelayMs)                     // slot is NOT held here
	}
}

maxConc := utils.GetEnvInt("NODE_RED_MAX_CONCURRENCY") // unset / 0 = no limit

Channel semaphore ရဲ့ ကန့်သတ်ချက်များ

  • Burst ကြီးရင် စောင့်ချိန် ရှည်ပါတယ်။ စောင့်ချိန် ကန့်သတ်ချက် မထားလို့ ticket 1000 ခု တစ်ပြိုင်နက် ဝင်ရင် နောက်ဆုံး ticket က (1000 ÷ 8) × 5s ≈ 10 မိနစ် စောင့်ပါတယ်။ ဒါပေမဲ့ retry job / FAILED_PERMANENT မဖြစ်ဘဲ အကုန် ပြီးပါတယ် (RETRY_WORKER_ENABLED=false နဲ့လည်း အဆင်ပြေ)။ Report မှာ စောင့်နေတဲ့ run တွေက FETCHING_CPE_STATUS မှာ ခဏ ပေါ်ပါမယ်။
  • Restart ဖြစ်ရင် စောင့်နေတဲ့ ticket ပျောက်ပါတယ်။ Channel က memory ထဲမှာပဲ ရှိလို့ slot စောင့်နေတုန်း / Node-RED ခေါ်နေတုန်း app restart ဖြစ်ရင် goroutine ပျောက်ပြီး run က RECEIVED / FETCHING_CPE_STATUS မှာ ပိတ်မိနေပါမယ်။ Semaphore ကြောင့် goroutine တွေ ပိုကြာကြာ စောင့်လို့ (စောင့်ချိန် ကန့်သတ်ချက် မရှိလို့ အထူးသဖြင့်) ဒီအန္တရာယ် ပိုကြီးပါတယ်။ ဒီ run တွေကို retry job အဖြစ် ပြန်ထည့်ပေးမယ့် sweeper လိုပါတယ် ("Node-RED Throughput Design" page အဆင့် ၁ · sweeper အသစ်)။
  • Instance တစ်ခုချင်း သီးသန့်ပါ။ Channel က process တစ်ခုထဲမှာပဲ ရှိလို့ noc instance 2 ခုဆိုရင် 8 + 8 = 16 ဖြစ်ပြီး Node-RED ရဲ့ 10 ကို ကျော်ပါတယ်။ Instance အရေအတွက်နဲ့ စားထားပါ (2 ခု → 4)၊ မဟုတ်ရင် Redis semaphore ကို သုံးပါ။
  • ရောက်တဲ့အစဉ် (FIFO) အာမခံချက် မရှိပါ။ Slot လွတ်တဲ့အခါ စောင့်နေသူထဲက ဘယ်သူ ရမလဲ Go runtime က ဆုံးဖြတ်ပါတယ်။ Ticket တစ်ရက် 1000 အတွက် ပြဿနာ မဟုတ်ပါ။
  • စောင့်နေတဲ့ goroutine တွေ memory ထဲ ရှိနေပါတယ်။ Goroutine တစ်ခု ~KB အနည်းငယ်ပဲ ဖြစ်လို့ ရာဂဏန်းအထိ ပြဿနာ မရှိပါ။
  • Node-RED limit က rate (တစ်စက္ကန့် 10 ခု) ဆိုရင် channel အစား golang.org/x/time/rate ကို သုံးရပါမယ်။ Node-RED team နဲ့ "တစ်ပြိုင်နက် 10" လား "တစ်စက္ကန့် 10" လား အတည်ပြုပါ။ ထည့်ရမယ့်နေရာကတော့ အတူတူပါ။
